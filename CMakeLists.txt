cmake_minimum_required(VERSION 3.28)
project(Briefcase.PregameTimer VERSION 0.3.0 LANGUAGES CXX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/mod-build.json" _build_config)
string(JSON _sdk_version GET "${_build_config}" sdkVersion)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
find_package(BriefcaseNativeSDK ${_sdk_version} EXACT CONFIG REQUIRED)
add_library(Briefcase.PregameTimer SHARED PregameTimer.cpp PregameTimerMod.cpp)
target_link_libraries(Briefcase.PregameTimer PRIVATE Briefcase::ModApi Briefcase::Json)
set_target_properties(Briefcase.PregameTimer PROPERTIES PREFIX "")
include(CTest)
if(BUILD_TESTING)
 add_executable(ModContracts tests/Contracts.cpp)
 target_link_libraries(ModContracts PRIVATE Briefcase::ModApi Briefcase::Json)
 add_test(NAME ModContracts COMMAND ModContracts)
 add_executable(ModLifecycle tests/Lifecycle.cpp)
 target_link_libraries(ModLifecycle PRIVATE Briefcase::ModApi Briefcase::Json ${CMAKE_DL_LIBS})
 target_compile_definitions(ModLifecycle PRIVATE MOD_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
 add_dependencies(ModLifecycle Briefcase.PregameTimer)
 add_test(NAME ModLifecycle COMMAND ModLifecycle)
 set_tests_properties(ModLifecycle PROPERTIES WORKING_DIRECTORY "$<TARGET_FILE_DIR:Briefcase.PregameTimer>" TIMEOUT 20)
endif()
install(TARGETS Briefcase.PregameTimer LIBRARY DESTINATION Briefcase/Mods/briefcase.pregame-timer RUNTIME DESTINATION Briefcase/Mods/briefcase.pregame-timer)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 file(READ "${CMAKE_CURRENT_SOURCE_DIR}/briefcase.mod.json" manifest)
 string(REPLACE ".dll" ".so" manifest "${manifest}")
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/briefcase.mod.json" "${manifest}")
 set(mod_manifest "${CMAKE_CURRENT_BINARY_DIR}/briefcase.mod.json")
else()
 set(mod_manifest "${CMAKE_CURRENT_SOURCE_DIR}/briefcase.mod.json")
endif()
install(FILES "${mod_manifest}" DESTINATION Briefcase/Mods/briefcase.pregame-timer)
install(FILES Data/config.json DESTINATION Briefcase/Mods/briefcase.pregame-timer/Data)
